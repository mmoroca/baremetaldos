# cmdline.txt - Circle's own boot options (NOT the Linux kernel command line).
#
# Circle reads this file itself; everything is "key=value" on one or more
# lines.  Documented in Circle's doc/cmdline.txt.
#
# Leave loglevel at 4 (notice) while bringing the port up: the emulator logs
# USB enumeration, the sound device it picked, the display mode and every
# DOSBox error through it, and the serial port is the only output that keeps
# working once the frame buffer owns the screen.
loglevel=4

# Log to screen *and* serial.  tty1 is the screen, tty2 is the serial port; use
# a single value if you only want one.  The serial port is worth having during
# development: connect at 115200 8N1.
logdev=tty1
#logdev=tty2

# Do not let the firmware's "rainbow" splash hold the display.
# (Handled by config.txt on modern firmware, kept here as a reminder.)

# Keymap for Circle's *own* console (before DOSBox starts).  DOSBox applies its
# own layout through KEYB in dosbox.conf; the two are independent.
keymap=es

# USB: some hubs and keyboards need a little longer before enumeration starts.
usbpowerdelay=500

# Clock.  The port runs its own governor (src/platform/bare_sysinfo.cpp): 600
# MHz at the DOS prompt and for light games, 1500 for DOSBox's default cycles,
# 2000 only when the emulated frames do not fit in their budget.  Nothing here
# is needed for that, and "fast=true" would only fight it - that option is read
# by Circle's CCPUThrottle, which this port deliberately does not instantiate.
#fast=true

# The temperature at which the governor drops one level, in degrees Celsius
# (40-85).  Default 80.  This is *earlier* than the firmware's own limit (85 C),
# on purpose: the point is to back off before the firmware has to.  Raise it if
# your board has a fan and you want the speed; lower it on a Pi in a small case.
socmaxtemp=80
